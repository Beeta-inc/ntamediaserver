POST

/upload

multipart/form-data

Universal file upload endpoint with automatic timestamp-based collision avoidance. Sanitizes filename and organizes storage by designated folder.

Request Parameters

Field Type Required Description
file Binary File Yes The raw file binary (JPEG, PNG, MP4, MP3, PDF, etc.)
type String Optional Destination category: feed, chat, videos, or docs. Defaults to chat.

async function uploadMedia(file, type = 'feed') {
  const formData = new FormData();
  formData.append('file', file);
  formData.append('type', type);

  const res = await fetch('https://ntamediaserver.pages.dev/upload', {
    method: 'POST',
    body: formData
  });

  if (!res.ok) throw new Error(`Upload failed: ${res.statusText}`);
  const data = await res.json();
  
  // Full public URL to embed in img/video/audio
  const publicUrl = `https://ntamediaserver.pages.dev${data.fileUrl}`;
  return { ...data, publicUrl };
}
import requests

def upload_file(file_path: str, upload_type: str = "feed"):
    url = "https://ntamediaserver.pages.dev/upload"
    with open(file_path, "rb") as f:
        files = {"file": f}
        data = {"type": upload_type}
        response = requests.post(url, files=files, data=data)
        response.raise_for_status()
        result = response.json()
        result["public_url"] = f"https://ntamediaserver.pages.dev{result['fileUrl']}"
        return result
import 'dart:io';
import 'package:http/http.dart' as http;
import 'dart:convert';

Future<String> uploadMedia(File file, {String type = 'feed'}) async {
  final uri = Uri.parse('https://ntamediaserver.pages.dev/upload');
  final request = http.MultipartRequest('POST', uri)
    ..fields['type'] = type
    ..files.add(await http.MultipartFile.fromPath('file', file.path));

  final streamedResponse = await request.send();
  final response = await http.Response.fromStream(streamedResponse);
  
  if (response.statusCode == 200) {
    final data = jsonDecode(response.body);
    return 'https://ntamediaserver.pages.dev${data['fileUrl']}';
  } else {
    throw Exception('Failed to upload file');
  }
}
import okhttp3.*
import java.io.File

fun uploadMedia(file: File, type: String = "feed"): String {
    val client = OkHttpClient()
    val requestBody = MultipartBody.Builder()
        .setType(MultipartBody.FORM)
        .addFormDataPart("type", type)
        .addFormDataPart(
            "file", file.name,
            RequestBody.create(MediaType.parse("application/octet-stream"), file)
        )
        .build()

    val request = Request.Builder()
        .url("https://ntamediaserver.pages.dev/upload")
        .post(requestBody)
        .build()

    client.newCall(request).execute().use { response ->
        if (!response.isSuccessful) throw Exception("Unexpected code $response")
        return response.body()?.string() ?: ""
    }
}
import Foundation

func uploadMedia(fileData: Data, fileName: String, type: String = "feed") async throws -> String {
    let url = URL(string: "https://ntamediaserver.pages.dev/upload")!
    var request = URLRequest(url: url)
    request.httpMethod = "POST"
    
    let boundary = UUID().uuidString
    request.setValue("multipart/form-data; boundary=\(boundary)", forHTTPHeaderField: "Content-Type")
    
    var body = Data()
    body.append("--\(boundary)\r\n".data(using: .utf8)!)
    body.append("Content-Disposition: form-data; name=\"type\"\r\n\r\n\(type)\r\n".data(using: .utf8)!)
    body.append("--\(boundary)\r\n".data(using: .utf8)!)
    body.append("Content-Disposition: form-data; name=\"file\"; filename=\"\(fileName)\"\r\n".data(using: .utf8)!)
    body.append("Content-Type: application/octet-stream\r\n\r\n".data(using: .utf8)!)
    body.append(fileData)
    body.append("\r\n--\(boundary)--\r\n".data(using: .utf8)!)
    
    let (data, response) = try await URLSession.shared.upload(for: request, from: body)
    return String(decoding: data, as: UTF8.self)
}

/media/{folder}/{filename}

Range Seeking & P2P Fallback

Streams uploaded media directly to clients. Supports HTTP 206 Partial Content (range requests) for video scrubbing, audio streaming, and image rendering. If a file is not on the primary device, it automatically executes a P2P fallback download from peer nodes without client disruption.

Deletes a media file locally from the active storage node and automatically broadcasts/propagates the deletion to all peer nodes across the cluster.

async function deleteMedia(folder, filename) {
  const res = await fetch(`https://ntamediaserver.pages.dev/media/${folder}/${filename}`, {
    method: 'DELETE'
  });
  const data = await res.json();
  if (!res.ok && res.status !== 404) throw new Error(data.error || 'Delete failed');
  return data; // { message: "File deleted successfully" }
}

React / Next.js Drop-in Component

Copy-paste this production-ready React component for seamless media uploads in your frontend:

MediaUploader.jsx 
import React, { useState } from 'react';

export function MediaUploader({ onUploaded, folder = 'feed' }) {
  const [uploading, setUploading] = useState(false);
  const [url, setUrl] = useState(null);

  const handleUpload = async (e) => {
    const file = e.target.files?.[0];
    if (!file) return;

    setUploading(true);
    try {
      const formData = new FormData();
      formData.append('file', file);
      formData.append('type', folder);

      const res = await fetch('https://ntamediaserver.pages.dev/upload', {
        method: 'POST',
        body: formData,
      });

      const data = await res.json();
      const fullUrl = `https://ntamediaserver.pages.dev${data.fileUrl}`;
      setUrl(fullUrl);
      if (onUploaded) onUploaded(fullUrl);
    } catch (err) {
      alert('Upload failed: ' + err.message);
    } finally {
      setUploading(false);
    }
  };

  return (
    <div className="p-4 border rounded-xl">
      <input type="file" onChange={handleUpload} disabled={uploading} />
      {uploading && <p className="text-sm mt-2 text-cyan-400">Uploading...</p>}
      {url && (
        <div className="mt-3">
          <img src={url} alt="Uploaded" className="max-h-48 rounded" />
        </div>
      )}
    </div>
  );
}
